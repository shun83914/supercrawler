### 示例 1：LangChain Python（推荐，最灵活）
# 思路：拉取 /api/skills/manifest → 用 pydantic.create_model 动态把 inputSchema 翻译成 BaseModel → 用 StructuredTool 注册给 agent。

# pip install langchain langchain-openai requests pydantic
import os, json, requests
from typing import Any, Dict, List, Type
from pydantic import BaseModel, Field, create_model
from langchain.tools import StructuredTool
from langchain_openai import ChatOpenAI
from langchain.agents import AgentExecutor, create_openai_functions_agent
from langchain_core.prompts import ChatPromptTemplate, MessagesPlaceholder

BASE_URL = os.getenv("SUPERCRAWLER_URL", "http://localhost:3000")
API_KEY  = os.getenv("SUPERCRAWLER_TOKEN", "")
HEADERS  = {"X-API-Key": API_KEY, "Content-Type": "application/json"}

JSON_TYPE_MAP = {"string": str, "integer": int, "number": float,
                 "boolean": bool, "array": list, "object": dict}

def schema_to_model(name: str, schema: Dict[str, Any]) -> Type[BaseModel]:
    """JSON Schema (draft-07 子集) → pydantic model"""
    props = schema.get("properties", {}) or {}
    required = set(schema.get("required", []) or [])
    fields = {}
    for k, v in props.items():
        py_t = JSON_TYPE_MAP.get(v.get("type", "string"), str)
        default = ... if k in required else v.get("default", None)
        fields[k] = (py_t, Field(default=default, description=v.get("description", "")))
    return create_model(name, **fields)

def call_supercrawler(method: str, path: str, payload: Dict[str, Any]) -> Dict[str, Any]:
    url = BASE_URL + path
    r = requests.request(method, url, headers=HEADERS,
                         json=payload if method == "POST" else None,
                         params=payload if method == "GET" else None,
                         timeout=120)
    body = r.json()
    if not body.get("success"):
        # 把业务错误码冒泡给 agent，让模型按 README 第6节决策
        raise RuntimeError(f"[{body.get('code')}] {body.get('message')} traceId={body.get('traceId')}")
    return body["data"]

def build_tools() -> List[StructuredTool]:
    manifest = requests.get(f"{BASE_URL}/api/skills/manifest", headers=HEADERS, timeout=10).json()
    # 注意：响应壳 data 字段才是真正 manifest 数组
    skills = manifest["data"] if isinstance(manifest, dict) and "data" in manifest else manifest
    tools = []
    for skill in skills:
        ArgsModel = schema_to_model(skill["name"].replace(".", "_") + "Args",
                                    skill["inputSchema"])
        method, path = skill["http"]["method"], skill["http"]["path"]
        # 闭包绑定，避免循环引用
        def make_fn(m, p):
            return lambda **kwargs: call_supercrawler(m, p, kwargs)
        tools.append(StructuredTool.from_function(
            func=make_fn(method, path),
            name=skill["name"].replace(".", "_"),
            description=f"{skill['description']}\nReturns: {skill['outputHint']}",
            args_schema=ArgsModel,
        ))
    return tools

# ---- Agent 装配 ----
tools = build_tools()
llm = ChatOpenAI(model="gpt-4o-mini", temperature=0)
prompt = ChatPromptTemplate.from_messages([
    ("system", "你是小红书数据分析助手。先调 health 看登录状态，"
               "默认不要 includeRecords，拿 file 路径后再 storage_peek。"
               "失败码 LOGIN_REQUIRED→调 auth_login；RATE_LIMITED→退避5分钟。"),
    ("user", "{input}"),
    MessagesPlaceholder("agent_scratchpad"),
])
agent = create_openai_functions_agent(llm, tools, prompt)
executor = AgentExecutor(agent=agent, tools=tools, verbose=True, max_iterations=8)

if __name__ == "__main__":
    print(executor.invoke({
        "input": "搜索『跑鞋』2025年6月以后、点赞≥1000的热门笔记5条，给我标题和发布时间"
    }))

python agent.py

# Dify 的"工具"模块原生支持 OpenAPI 3.0 schema 导入，需要先把 manifest 转成 OpenAPI。
# Step 1：用一个小脚本把 manifest 转 OpenAPI（一次性）
# manifest_to_openapi.py
import json, requests, yaml

BASE = "http://localhost:3000"
manifest_resp = requests.get(f"{BASE}/api/skills/manifest").json()
skills = manifest_resp.get("data", manifest_resp)

paths = {}
for s in skills:
    method = s["http"]["method"].lower()
    path = s["http"]["path"]
    op = {
        "operationId": s["name"].replace(".", "_"),
        "summary": s["description"][:120],
        "description": s["description"],
        "responses": {"200": {"description": s["outputHint"]}},
    }
    if method == "post":
        op["requestBody"] = {"required": True, "content": {
            "application/json": {"schema": s["inputSchema"]}}}
    else:  # GET
        op["parameters"] = [
            {"name": k, "in": "query",
             "required": k in (s["inputSchema"].get("required") or []),
             "schema": v}
            for k, v in (s["inputSchema"].get("properties") or {}).items()
        ]
    paths.setdefault(path, {})[method] = op

openapi = {
    "openapi": "3.0.0",
    "info": {"title": "SuperCrawler", "version": "0.1.0"},
    "servers": [{"url": BASE}],
    "components": {"securitySchemes": {
        "ApiKey": {"type": "apiKey", "in": "header", "name": "X-API-Key"}
    }},
    "security": [{"ApiKey": []}],
    "paths": paths,
}
with open("supercrawler.openapi.yaml", "w") as f:
    yaml.safe_dump(openapi, f, allow_unicode=True, sort_keys=False)
print("written supercrawler.openapi.yaml")

### 示例 3：OpenAI Function Calling（Python 原生，最轻量）
# 不依赖 LangChain 框架，直接调 OpenAI Tools API：
# pip install openai requests
import os, json, requests
from openai import OpenAI

BASE = "http://localhost:3000"
HEADERS = {"X-API-Key": os.getenv("SUPERCRAWLER_TOKEN", ""),
           "Content-Type": "application/json"}
client = OpenAI()

def fetch_tools():
    resp = requests.get(f"{BASE}/api/skills/manifest", headers=HEADERS).json()
    skills = resp.get("data", resp)
    tools, route = [], {}
    for s in skills:
        name = s["name"].replace(".", "_")
        tools.append({
            "type": "function",
            "function": {
                "name": name,
                "description": f"{s['description']}\nReturns: {s['outputHint']}",
                "parameters": s["inputSchema"],
            },
        })
        route[name] = (s["http"]["method"], s["http"]["path"])
    return tools, route

def call_route(method, path, args):
    r = requests.request(method, BASE + path, headers=HEADERS,
                         json=args if method == "POST" else None,
                         params=args if method == "GET" else None, timeout=120)
    return r.json()  # 直接把响应壳整体返回给模型，让它读 success/code

def run(user_msg: str):
    tools, route = fetch_tools()
    messages = [
        {"role": "system",
         "content": "你是小红书数据分析助手。失败码 LOGIN_REQUIRED→调 auth_login，"
                    "RATE_LIMITED→停止；默认不要 includeRecords。"},
        {"role": "user", "content": user_msg},
    ]
    for _ in range(8):
        resp = client.chat.completions.create(
            model="gpt-4o-mini", messages=messages, tools=tools, tool_choice="auto")
        msg = resp.choices[0].message
        messages.append(msg)
        if not msg.tool_calls:
            print("\nFINAL:", msg.content); return
        for tc in msg.tool_calls:
            method, path = route[tc.function.name]
            args = json.loads(tc.function.arguments or "{}")
            result = call_route(method, path, args)
            messages.append({"role": "tool", "tool_call_id": tc.id,
                             "content": json.dumps(result, ensure_ascii=False)[:8000]})
    print("达到迭代上限")

if __name__ == "__main__":
    run("帮我抓『露营装备』搜索结果热门 3 条，告诉我标题和点赞数")

### 1.3 Agent 侧 Prompt 模板（建议放进 agent 的 system prompt）
你具备 supercrawler 工具（小红书数据抓取）。调用规范：

1. 任何 xhs_* 调用前，先 health 检查 accounts.profilesOnDisk 与 onlineContexts
2. 若 auth_status 返回 loggedIn=false → 调 auth_login（需宿主机 headful 环境）

4. 批量目标用 xhs_batch 而非自己 for-loop（服务端自带延迟防风控）

   - XHS_TARGET_NOT_FOUND → 放弃该目标
   - TIMEOUT / NAVIGATION_FAILED → 重试 ≤2 次

## 工作流（严格按顺序）

### Step 1: 前置检查
- 调用 `supercrawler:health`，确认服务在线
- 调用 `supercrawler:auth_status` 探测 accountId（默认 `default`）
- 若 `loggedIn=false` → 提示用户：**"需要扫码登录，请在服务宿主机执行 auth_login"**
  并调用 `supercrawler:auth_login`（宿主机会自动弹出浏览器）

### Step 2: 识别任务类型并调对应工具

| 用户意图 | 工具 | 关键参数 |
|---|---|---|
| 抓指定笔记详情 | `xhs_scrape_notes` | `noteIds: string[]` |
| 抓用户主页 | `xhs_scrape_user` | `userId`, `noteLimit` |
| 关键词搜索 | `xhs_scrape_search` | `keywords[], sort, limit, minLikes, noteType, publishedAfter/Before` |
| 抓笔记评论 | `xhs_scrape_comments` | `noteId, limit` |
| 混合批量（≥3 目标） | `xhs_batch` | `tasks: [{type, id}]` |

### Step 3: 取明细
- **默认不要** 设置 `includeRecords=true`
- 拿到 `data.file` 路径后，用 `storage_peek` 分页读（`limit=50`）

### Step 4: 错误码决策
| code | 处理 |
|---|---|
| `OK` | 正常返回 |
| `LOGIN_REQUIRED` / `LOGIN_TIMEOUT` | 调 `auth_login` |
| `RATE_LIMITED` / `XHS_BLOCKED` | 停止，告知用户退避 ≥5 分钟 |
| `XHS_TARGET_NOT_FOUND` | 该目标跳过 |
| `TIMEOUT` / `NAVIGATION_FAILED` | 同参重试 ≤2 次 |

## 示例对话

> User: 搜"跑鞋"2025年6月后热门、点赞≥1000的前 5 条

## 硬性约束
- **不要并发调用多个 xhs_* 工具**（服务端强制 concurrency=1，并发只会排队浪费上下文）
- **不要在 keywords 数组里塞 >10 个**（单次超过会被风控）
- **不要自己拼 URL 或发 raw HTTP**，一切走 MCP 工具

# 如果你的 OpenClaw 支持 TypeScript skill 脚本（index.ts 配合 skill.json），可以加一份做自动预检：~/.openclaw/skills/xhs-scraper/index.ts
# // 自动在每次调用前检查服务是否存活，挂了就尝试拉起
import { spawn } from "node:child_process";
import { resolve } from "node:path";

const SUPERCRAWLER_DIR = process.env.SUPERCRAWLER_DIR ??
  "/Users/suhao/Documents/coding/supercrawler";

export async function onBeforeInvoke(toolName: string): Promise<void> {
  if (!toolName.startsWith("supercrawler:")) return;

  try {
    const r = await fetch("http://localhost:3000/api/health", {
      headers: { "X-API-Key": process.env.SUPERCRAWLER_TOKEN ?? "" },
      signal: AbortSignal.timeout(1500),
    });
    if (r.ok) return;
  } catch {
    // 服务未起 → 后台拉起 HTTP 模式方便 storage_peek 走 HTTP
    spawn("node", [resolve(SUPERCRAWLER_DIR, "dist/main.js")], {
      cwd: SUPERCRAWLER_DIR,
      detached: true,
      stdio: "ignore",
      env: { ...process.env, CLOAK_HEADLESS: "true" },
    }).unref();
    await new Promise((r) => setTimeout(r, 3000));
  }
}

export async function onAfterInvoke(toolName: string, result: unknown): Promise<void> {
  // 关键失败码预警日志
  const r = result as { isError?: boolean; content?: Array<{ text: string }> };
  if (!r?.isError) return;
  const txt = r.content?.[0]?.text ?? "";
  if (/RATE_LIMITED|XHS_BLOCKED/.test(txt)) {
    console.error(`[xhs-scraper] ⚠️  风控触发，建议退避 5+ 分钟：${txt.slice(0, 200)}`);
  }
}
